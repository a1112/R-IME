find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent)
set(CMAKE_AUTOMOC ON)
add_library(rime-ime-window-effects STATIC windoweffects.cpp windoweffects.h effectspolicy.h macosutils.h)
target_include_directories(rime-ime-window-effects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(rime-ime-window-effects PUBLIC Qt6::Core Qt6::Gui)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(rime-ime-window-effects PRIVATE macosutils.mm)
  target_link_libraries(rime-ime-window-effects PRIVATE "-framework AppKit")
elseif(WIN32)
  target_link_libraries(rime-ime-window-effects PRIVATE dwmapi)
endif()
add_library(rime-ime-settings-backend STATIC backend.cpp backend.hpp)
target_include_directories(rime-ime-settings-backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/src/ipc)
target_link_libraries(rime-ime-settings-backend PUBLIC Qt6::Core Qt6::Concurrent rime-ime-core rime-ime-ipc)
add_executable(r-ime-settings main.cpp)
if(WIN32)
  find_package(Qt6 6.11 REQUIRED COMPONENTS QuickControls2)
  set_target_properties(r-ime-settings PROPERTIES WIN32_EXECUTABLE TRUE)
  target_sources(r-ime-settings PRIVATE ../windows/resources.rc)
  target_include_directories(r-ime-settings PRIVATE ${PROJECT_SOURCE_DIR}/assets)
endif()
qt_add_resources(r-ime-settings settings-resources PREFIX "/settings" FILES Main.qml WindowChrome.qml ResizeHandles.qml)
if(WIN32)
  qt_add_resources(r-ime-settings windows-settings-resources PREFIX "/settings" FILES Windows.qml)
endif()
qt_add_resources(r-ime-settings branding PREFIX "/branding" BASE "${PROJECT_SOURCE_DIR}/assets" FILES "${PROJECT_SOURCE_DIR}/assets/r-ime.svg")
target_link_libraries(r-ime-settings PRIVATE rime-ime-settings-backend rime-ime-window-effects Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS r-ime-settings RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING)

 add_test(NAME settings-chrome COMMAND settings-chrome-test)
 set_tests_properties(settings-chrome PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60)
 add_executable(settings-test ${PROJECT_SOURCE_DIR}/tests/settings_test.cpp)
 target_link_libraries(settings-test PRIVATE rime-ime-settings-backend Qt6::Test)
 add_test(NAME settings-backend COMMAND settings-test)
 add_test(NAME settings-qml-smoke COMMAND r-ime-settings --smoke-test)
 set_tests_properties(settings-qml-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;LANG=C.UTF-8" TIMEOUT 60)
 if(RIME_BUILD_ENGINE)
