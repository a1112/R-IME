if(WIN32)
  set(RIME_SDK_ROOT "" CACHE PATH "Official librime Windows dist directory")
  find_path(RIME_INCLUDE rime_api.h PATHS "${RIME_SDK_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
  find_library(RIME_LIBRARY rime PATHS "${RIME_SDK_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
  add_library(rime-native SHARED IMPORTED)
  set_target_properties(rime-native PROPERTIES IMPORTED_IMPLIB "${RIME_LIBRARY}" IMPORTED_LOCATION "${RIME_SDK_ROOT}/lib/rime.dll" INTERFACE_INCLUDE_DIRECTORIES "${RIME_INCLUDE}")
  set(RIME_TARGET rime-native)
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(yaml-cpp URL https://github.com/jbeder/yaml-cpp/archive/refs/tags/0.8.0.tar.gz URL_HASH SHA256=fbe74bbdcee21d656715688706da3c8becfd946d92cd44705cc6098bb23b3a16 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(yaml-cpp)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RIME REQUIRED IMPORTED_TARGET rime>=1.7)
  find_package(yaml-cpp REQUIRED)
  set(RIME_TARGET PkgConfig::RIME)
endif()
add_library(rime-ime-engine STATIC engine.cpp)
target_compile_features(rime-ime-engine PUBLIC cxx_std_20)
target_link_libraries(rime-ime-engine PUBLIC rime-ime-core ${RIME_TARGET} yaml-cpp)
if(WIN32)
  add_executable(r-ime-engine ../windows/worker.cpp)
  if(MSVC)
    set_target_properties(r-ime-engine PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(r-ime-engine PRIVATE /ENTRY:wmainCRTStartup)
  endif()
  target_link_libraries(r-ime-engine PRIVATE rime-ime-ipc advapi32)
  add_custom_command(TARGET r-ime-engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RIME_SDK_ROOT}/lib/rime.dll" "$<TARGET_FILE_DIR:r-ime-engine>")
else()
  add_executable(r-ime-engine main.cpp)
endif()
target_link_libraries(r-ime-engine PRIVATE rime-ime-engine Threads::Threads)
target_compile_definitions(r-ime-engine PRIVATE RIME_IME_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/r-ime/rime")
install(TARGETS r-ime-engine RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
