set(protocol_sources)
function(rime_protocol name xml)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/wayland-${name}-client-protocol.h")
  set(code "${CMAKE_CURRENT_BINARY_DIR}/wayland-${name}-protocol.c")
  add_custom_command(OUTPUT "${header}" COMMAND ${WAYLAND_SCANNER} client-header "${xml}" "${header}" DEPENDS "${xml}")
  add_custom_command(OUTPUT "${code}" COMMAND ${WAYLAND_SCANNER} private-code "${xml}" "${code}" DEPENDS "${xml}")
  set(protocol_sources ${protocol_sources} "${header}" "${code}" PARENT_SCOPE)
endfunction()
