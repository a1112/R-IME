cmake_minimum_required(VERSION 3.22)
project(RIME VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
configure_file(cmake/uninstall.cmake.in uninstall.cmake @ONLY)
add_custom_target(uninstall COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake)
option(RIME_BUILD_ENGINE "Build the isolated librime engine" ON)
set(RIME_LINUX_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(RIME_LINUX_DEFAULT ON)
endif()
option(RIME_BUILD_FCITX "Build the Linux Fcitx 5 input method" ${RIME_LINUX_DEFAULT})
option(RIME_BUILD_SETTINGS "Build Qt Quick settings" ON)
option(RIME_BUILD_WINDOWS_TSF "Build Windows TSF adapter" ${WIN32})
option(RIME_BUILD_CONTROL "Build the local control API and command-line tool" ON)
if(MSVC)
  add_compile_options(/utf-8 /W3)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
endif()
if(WIN32)
  include(FetchContent)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  set(JSON_Install ${RIME_BUILD_CONTROL} CACHE BOOL "Install JSON dependency for the control SDK" FORCE)
  FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/archive/refs/tags/v3.11.3.tar.gz URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(nlohmann_json)
else()
  find_package(nlohmann_json 3.10 REQUIRED)
endif()
find_package(Threads REQUIRED)
add_subdirectory(src/core)
if(WIN32)
  add_library(rime-ime-ipc STATIC src/ipc/client_windows.cpp)
  target_link_libraries(rime-ime-ipc PUBLIC rime-ime-core advapi32)
  target_include_directories(rime-ime-ipc PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/src/ipc>)
else()
  add_library(rime-ime-ipc STATIC src/ipc/client.cpp)
  target_link_libraries(rime-ime-ipc PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
endif()
if(RIME_BUILD_CONTROL)
  add_subdirectory(src/control)
  include(CMakePackageConfigHelpers)
  set_target_properties(rime-ime-core PROPERTIES EXPORT_NAME core)
  set_target_properties(rime-ime-ipc PROPERTIES EXPORT_NAME ipc)
  set_target_properties(rime-ime-control PROPERTIES EXPORT_NAME control)
  install(TARGETS rime-ime-core rime-ime-ipc rime-ime-control
    EXPORT RIMETargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(TARGETS r-ime-ctl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES include/rime_ime/config.hpp include/rime_ime/control.hpp
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/rime_ime)
  install(FILES docs/control-api.md DESTINATION ${CMAKE_INSTALL_DATADIR}/r-ime/docs)
  install(FILES examples/control_client.py examples/control_client.cpp examples/CMakeLists.txt
    DESTINATION ${CMAKE_INSTALL_DATADIR}/r-ime/examples)
  install(EXPORT RIMETargets NAMESPACE RIME:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RIME)
  configure_package_config_file(cmake/RIMEConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/RIMEConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RIME)
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/RIMEConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/RIMEConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/RIMEConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RIME)
endif()
if(WIN32 AND RIME_BUILD_WINDOWS_TSF)
  add_subdirectory(src/windows/tsf)
endif()
if(RIME_BUILD_FCITX)
  add_subdirectory(src/panel)
endif()
if(RIME_BUILD_ENGINE)
  add_subdirectory(src/engine)
endif()
if(RIME_BUILD_FCITX)
  add_subdirectory(src/fcitx)
endif()
if(RIME_BUILD_SETTINGS)
  add_subdirectory(src/settings)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(NOT WIN32)
install(FILES packaging/org.r-ime.Settings.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/r-ime.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(DIRECTORY data/rime/ DESTINATION ${CMAKE_INSTALL_DATADIR}/r-ime/rime)
